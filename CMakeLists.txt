cmake_minimum_required(VERSION 3.21)
project(phi4 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(phi4_core INTERFACE)
target_include_directories(phi4_core INTERFACE include)
if(MSVC)
  target_compile_options(phi4_core INTERFACE /W4)
else()
  target_compile_options(phi4_core INTERFACE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_executable(test_model tests/cpp/test_model.cpp)
target_link_libraries(test_model PRIVATE phi4_core)
add_test(NAME model COMMAND test_model)
add_executable(phi4 src/main.cpp)
target_link_libraries(phi4 PRIVATE phi4_core)
target_compile_definitions(phi4 PRIVATE
    PHI4_BUILD_TYPE="${CMAKE_BUILD_TYPE}"
    PHI4_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
add_executable(test_distribution tests/cpp/test_distribution.cpp)
target_link_libraries(test_distribution PRIVATE phi4_core)
add_test(NAME distribution COMMAND test_distribution)
set_tests_properties(distribution PROPERTIES LABELS "distribution")
add_test(NAME workflow COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/python/test_workflow.py" "$<TARGET_FILE:phi4>")
add_test(NAME statistics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/python/test_statistics.py")
add_test(NAME analysis COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/python/test_analysis.py")
